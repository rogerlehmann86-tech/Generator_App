 if(!valid.length)return;

   id:crypto.randomUUID(),
   name:p.name||'Gerät',
   watt:Number(p.watt)||0,
   pf:Number(p.power_factor)||1,
   startMult:Number(p.start_multiplier)||1,
   useFactors:!!p.use_factors,
   icon_key:p.icon_key||'plug',

 }));

   button.classList.add('bulk-added');

   setTimeout(()=>{button.classList.remove('bulk-added');button.innerHTML=original},1400);
 }
}
