function computeSummary(){
  const totalVA=devices.reduce((sum,d)=>sum+(d.useFactors?(Number(d.watt)/(Number(d.pf)||1))*(Number(d.startMult)||1):Number(d.watt)),0);
  const totalKVA=totalVA/1000;const needs400V=devices.some(d=>d.voltage==='400V');const steps=[.65,.8,1,1.5,2,2.5,3,4,5,6.5,8,10,12,15,20,25,30,40,50,60,80,100];const marketKVA=steps.find(s=>s>=totalKVA)??Math.ceil(totalKVA/10)*10;
  const rentals=[...rentalGenerators].sort((a,b)=>Number(a.kva)-Number(b.kva));const rental=rentals.find(r=>Number(r.kva)>=totalKVA&&(!needs400V||String(r.voltage_support).includes('400')))||null;
  return{totalKVA,marketKVA,needs400V,rental};
}
