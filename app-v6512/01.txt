      db.from('generator_applications').select('*').eq('active',true).order('sort_order'),
      db.from('generator_application_devices').select('*').order('sort_order')
    ]);
    if(cr.error)throw cr.error;if(dr.error)throw dr.error;if(rr.error)throw rr.error;if(ar.error)throw ar.error;if(alr.error)throw alr.error;
    categories=cr.data||[];presets=dr.data||[];applications=ar.data||[];applicationLinks=alr.data||[];
    rentalGenerators=(rr.data||[]).filter(x=>x.products?.active!==false).map(x=>({...x,name:x.products.name,slug:x.products.slug}));
    el('dataNotice').classList.add('hidden');
  }catch(e){
    console.warn('Generator-Stammdaten nicht erreichbar:',e);categories=fallbackCategories;presets=fallbackPresets;rentalGenerators=fallbackRentals;applications=fallbackApplications;applicationLinks=fallbackApplicationLinks;
    notice('Die zentralen Stammdaten sind noch nicht eingerichtet oder nicht erreichbar. Es werden vorläufig die eingebauten Standardwerte verwendet.',true);
  }
}
