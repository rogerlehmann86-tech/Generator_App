async function saveCategory(ev){ev.preventDefault();const id=el('categoryId').value||null,payload={name:el('categoryName').value.trim(),icon_key:el('categoryIcon').value||'th-large',sort_order:Number(el('categorySort').value)||100,active:el('categoryActive').checked};if(!payload.name)return show('Name ist erforderlich.',true);const q=id?db.from('generator_categories').update(payload).eq('id',id):db.from('generator_categories').insert(payload);const {error}=await q;if(error)return show(error.message,true);el('categoryDialog').close();show('Kategorie gespeichert.',false,true);await loadAll()}
async function saveDevice(ev){ev.preventDefault();const id=el('deviceId').value||null,payload={category_id:el('deviceCategory').value,name:el('deviceName').value.trim(),icon_key:el('deviceIcon').value||'plug',watt:Number(el('deviceWatt').value),voltage:el('deviceVoltage').value,power_factor:Number(el('devicePf').value),start_multiplier:Number(el('deviceStartMult').value),use_factors:el('deviceUseFactors').checked,sort_order:Number(el('deviceSort').value)||100,active:el('deviceActive').checked};if(!payload.name||!payload.category_id||payload.watt<=0||payload.power_factor<=0||payload.power_factor>1||payload.start_multiplier<1)return show('Bitte Leistungsdaten prüfen. cos φ: 0.1–1, Anlauffaktor: mindestens 1.',true);const q=id?db.from('generator_devices').update(payload).eq('id',id):db.from('generator_devices').insert(payload);const {error}=await q;if(error)return show(error.message,true);el('deviceDialog').close();show('Gerät gespeichert.',false,true);await loadAll()}
function openApplication(a=null){el('applicationDialogTitle').textContent=a?'Anwendung bearbeiten':'Anwendung hinzufügen';el('applicationId').value=a?.id||'';el('applicationName').value=a?.name||'';el('applicationDescription').value=a?.description||'';el('applicationIcon').innerHTML=iconOptions(a?.icon_key||'magic');el('applicationSort').value=a?.sort_order??100;el('applicationActive').checked=a?.active!==false;const linked=new Set(applicationLinks.filter(l=>String(l.application_id)===String(a?.id)).map(l=>String(l.device_id)));el('applicationDevicePicker').innerHTML=devices.filter(d=>d.active!==false).map(d=>`<label class="picker-item"><input type="checkbox" value="${d.id}" ${linked.has(String(d.id))?'checked':''}><i class="fa fa-${esc(d.icon_key||'plug')}"></i><span>${esc(d.name)}<small>${d.watt} W · ${esc(d.voltage)}</small></span></label>`).join('');el('applicationDialog').showModal()}window.editApplication=id=>openApplication(applications.find(x=>String(x.id)===String(id)));
async function saveApplication(ev){ev.preventDefault();const id=el('applicationId').value||null,payload={name:el('applicationName').value.trim(),description:el('applicationDescription').value.trim(),icon_key:el('applicationIcon').value||'magic',sort_order:Number(el('applicationSort').value)||100,active:el('applicationActive').checked};if(!payload.name)return show('Name ist erforderlich.',true);let appId=id;if(id){const {error}=await db.from('generator_applications').update(payload).eq('id',id);if(error)return show(error.message,true)}else{const {data,error}=await db.from('generator_applications').insert(payload).select('id').single();if(error)return show(error.message,true);appId=data.id}const selected=[...el('applicationDevicePicker').querySelectorAll('input:checked')].map((x,i)=>({application_id:appId,device_id:x.value,sort_order:(i+1)*10}));let q=await db.from('generator_application_devices').delete().eq('application_id',appId);if(q.error)return show(q.error.message,true);if(selected.length){q=await db.from('generator_application_devices').insert(selected);if(q.error)return show(q.error.message,true)}el('applicationDialog').close();show('Anwendung gespeichert.',false,true);await loadAll()}
